const makeRadar = (items) => {
  const wrap = el("div", "ac-radar-wrap");
  const canvas = el("canvas", "ac-radar");
  canvas.setAttribute("aria-label", "六维能力证据地图");
  canvas.setAttribute("aria-describedby", "ac-radar-status-list");
  const accessibleList = el("ul", "ac-visually-hidden");
  accessibleList.id = "ac-radar-status-list";
  items.forEach((item) => {
    accessibleList.append(el("li", "", `${text(item.dimension_name)}：${stateLabel[effectiveState(item)] ?? effectiveState(item)}；达标证据为${text(item.success_criteria)}`));
  });
  wrap.append(canvas, accessibleList);

  const draw = () => {
    const rect = canvas.getBoundingClientRect();
    if (!rect.width || !rect.height) return;
    const dpr = window.devicePixelRatio || 1;
    canvas.width = Math.round(rect.width * dpr);
    canvas.height = Math.round(rect.height * dpr);
    const ctx = canvas.getContext("2d");
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    ctx.clearRect(0, 0, rect.width, rect.height);
    const cx = rect.width / 2;
    const cy = rect.height / 2 + 4;
    const radius = Math.min(rect.width, rect.height) * .355;
    const angle = (index) => -Math.PI / 2 + Math.PI * 2 * index / 6;
    const point = (index, scale) => ({ x: cx + Math.cos(angle(index)) * radius * scale, y: cy + Math.sin(angle(index)) * radius * scale });
    for (let level = 1; level <= 5; level += 1) {
      ctx.beginPath();
      items.forEach((_, index) => {
        const p = point(index, level / 5);
        index ? ctx.lineTo(p.x, p.y) : ctx.moveTo(p.x, p.y);
      });
      ctx.closePath();
      ctx.strokeStyle = level === 5 ? "rgba(95,91,85,.4)" : "rgba(95,91,85,.22)";
      ctx.lineWidth = 1;
      ctx.stroke();
    }
    items.forEach((_, index) => {
      const p = point(index, 1);
      ctx.beginPath();
      ctx.moveTo(cx, cy);
      ctx.lineTo(p.x, p.y);
      ctx.strokeStyle = "rgba(95,91,85,.23)";
      ctx.stroke();
    });
    const points = items.map((item, index) => point(index, stateValue(effectiveState(item)) / 5));
    ctx.beginPath();
    points.forEach((p, index) => index ? ctx.lineTo(p.x, p.y) : ctx.moveTo(p.x, p.y));
    ctx.closePath();
    ctx.fillStyle = "rgba(93,126,167,.24)";
    ctx.fill();
    ctx.strokeStyle = "#2e69a5";
    ctx.lineWidth = 3;
    ctx.stroke();
    points.forEach((p) => {
      ctx.beginPath();
      ctx.arc(p.x, p.y, 4, 0, Math.PI * 2);
      ctx.fillStyle = "#f2ece2";
      ctx.fill();
      ctx.strokeStyle = "#2e69a5";
      ctx.lineWidth = 2;
      ctx.stroke();
    });
    ctx.fillStyle = "#3f3d39";
    ctx.font = '13px "Microsoft YaHei UI", "Microsoft YaHei", sans-serif';
    ctx.textBaseline = "middle";
    items.forEach((item, index) => {
      const p = point(index, 1.19);
      const horizontal = Math.cos(angle(index));
      ctx.textAlign = horizontal > .2 ? "left" : horizontal < -.2 ? "right" : "center";
      ctx.fillText(text(item.dimension_name), p.x, p.y, Math.max(64, rect.width * .26));
    });
  };
  requestAnimationFrame(draw);
  const observer = new ResizeObserver(draw);
  observer.observe(canvas);
  container.register?.(() => observer.disconnect());
  return wrap;
};

const openDimensionEditor = () => {

  }));
  const overlay = el("div", "ac-editor-overlay");
  const drawer = el("aside", "ac-editor-drawer");
  const previousFocus = document.activeElement;
  drawer.setAttribute("role", "dialog");
  drawer.setAttribute("aria-modal", "true");
  drawer.setAttribute("aria-labelledby", "ac-editor-title");
  overlay.append(drawer);
  document.body.append(overlay);
  container.inert = true;
  container.setAttribute("aria-hidden", "true");
  const close = () => {
    overlay.remove();
    container.inert = false;
    container.removeAttribute("aria-hidden");
    previousFocus?.focus?.();
  };
  const onEscape = (event) => { if (event.key === "Escape") { close(); document.removeEventListener("keydown", onEscape); } };
  document.addEventListener("keydown", onEscape);
  drawer.addEventListener("keydown", (event) => {
    if (event.key !== "Tab") return;
    const focusable = drawer.querySelectorAll('button:not(:disabled), input:not(:disabled), textarea:not(:disabled), [tabindex]:not([tabindex="-1"])');
    if (!focusable.length) return;
    const first = focusable[0];
    const last = focusable[focusable.length - 1];
    if (event.shiftKey && document.activeElement === first) {
      event.preventDefault();
      last.focus();
    } else if (!event.shiftKey && document.activeElement === last) {
      event.preventDefault();
      first.focus();
    }
  });

  const render = (preview = false) => {
    drawer.empty();
    const header = el("div", "ac-editor-header");
    const titleWrap = el("div", "");
    titleWrap.append(el("span", "ac-kicker", "我的六边形模板"), el("h2", "", preview ? "确认六维更改" : "编辑六维模板"));

    }

    drawer.append(el("p", "ac-editor-help", "固定六个位置。可修改名称、顺序和达标证据；能力等级只能经证据流程更新。"));

function isoWeekInfo(date) {
  const local = new Date(date.getFullYear(), date.getMonth(), date.getDate());
  const weekday = local.getDay() || 7;
  const monday = new Date(local);
  monday.setDate(local.getDate() - weekday + 1);
  const thursday = new Date(local);
  thursday.setDate(local.getDate() + 4 - weekday);
  const yearStart = new Date(thursday.getFullYear(), 0, 1);
  const week = Math.ceil((((thursday - yearStart) / 86400000) + 1) / 7);
  const sunday = new Date(monday);
  sunday.setDate(monday.getDate() + 6);
  const dateText = (value) => value.getFullYear() + "-" + pad2(value.getMonth() + 1) + "-" + pad2(value.getDate());
  return { key: thursday.getFullYear() + "-W" + pad2(week), start: dateText(monday), end: dateText(sunday) };
}

function parseGeneratedFrontmatter(text) {
  const match = String(text || "").match(/^---\r?\n([\s\S]*?)\r?\n---/);
  if (!match) return {};
  const result = {};
  const lines = match[1].split(/\r?\n/);
  for (let index = 0; index < lines.length; index += 1) {
    const line = lines[index];
    const colon = line.indexOf(":");
    if (colon < 1 || /^\s/.test(line)) continue;
    const key = line.slice(0, colon).trim();
    const raw = line.slice(colon + 1).trim();
    if (/^[|>][-+]?\s*$/.test(raw)) {
      const block = [];
      while (index + 1 < lines.length && (/^\s/.test(lines[index + 1]) || lines[index + 1] === "")) {
        index += 1;
        block.push(lines[index].replace(/^ {1,4}/, ""));
      }
      result[key] = raw.startsWith(">") ? block.join(" ").trim() : block.join("\n").replace(/\n+$/, "");
    } else if (raw === "null" || raw === "~") result[key] = null;
    else if (raw === "true" || raw === "false") result[key] = raw === "true";
    else if (/^-?(?:0|[1-9]\d*)(?:\.\d+)?$/.test(raw)) result[key] = Number(raw);
    else if (raw.startsWith('"')) {
      try { result[key] = JSON.parse(raw); } catch (error) { result[key] = raw; }
    } else result[key] = raw;
  }
  return result;
}
function parseFinancePayload(text) {
  const source = String(text || "");
  const blockStart = source.indexOf("<!-- finance-json:start -->");
  const blockEnd = source.indexOf("<!-- finance-json:end -->", blockStart + 1);
  if (blockStart < 0 || blockEnd < 0) return null;
  const fence = String.fromCharCode(96).repeat(3);
  const jsonStart = source.indexOf(fence + "json", blockStart);
  const jsonEnd = source.indexOf(fence, jsonStart + fence.length + 4);
  if (jsonStart < 0 || jsonEnd < 0 || jsonEnd > blockEnd) return null;
  const payload = source.slice(jsonStart + fence.length + 4, jsonEnd).trim();
  try { return JSON.parse(payload); } catch (error) { return null; }
}

function storedMoney(value) {

function deepValue(source, keys, depth = 0) {
  if (!source || typeof source !== "object" || depth > 4) return null;
  for (const key of keys) {
    if (source[key] !== undefined && source[key] !== null) return source[key];
  }
  for (const value of Object.values(source)) {
    if (!value || typeof value !== "object") continue;
    const found = deepValue(value, keys, depth + 1);
    if (found !== null && found !== undefined) return found;
  }
  return null;
}

function actionMetric(keys) {
  if (!state.actionPlan?.payload) return null;
  const resultValue = deepValue(state.actionPlan.payload.result, keys);
  if (resultValue !== null && resultValue !== undefined) return resultValue;
  return deepValue(state.actionPlan.payload.input, keys);
}

function actionGap(stage) {
  const envelope = state.actionPlan?.payload?.result || {};
  const result = envelope.result || envelope.preview || envelope;
  return finiteNumber(result?.gaps?.[stage]?.remaining);
}

function remainingMetric(target, directKeys, actualKeys) {
  const direct = finiteNumber(actionMetric(directKeys));
  if (direct !== null) return Math.max(0, direct);
  const targetValue = finiteNumber(target);
  const actual = finiteNumber(actionMetric(actualKeys));
  if (targetValue === null || actual === null) return null;
  return Math.max(0, targetValue - actual);
}

function classifyProject(project) {
  const quote = storedMoney(project.quote);
  const payment = storedMoney(project.payment);
  const receivable = storedMoney(project.acceptedReceivable);
  if (!project.client || !project.project || !project.next || !quote.valid || !payment.valid || !receivable.valid) return "clarify";
  if (receivable.value !== null && payment.value !== null && payment.value > receivable.value) return "clarify";
  if (receivable.value !== null && (payment.value === null || payment.value < receivable.value)) return "collection";
  if (!project.review && payment.value !== null) return "review";
  return "active";
}
function operatingTotals(active) {
  let quoteCents = 0;
  let paymentCents = 0;
  let outstandingCents = 0;
  let quoteCount = 0;
  let paymentCount = 0;
  let outstandingKnown = 0;
  let outstandingUnknown = 0;
  active.forEach((project) => {
    const quote = storedMoney(project.quote);
    const payment = storedMoney(project.payment);
    const receivable = storedMoney(project.acceptedReceivable);
    if (quote.valid && quote.value !== null) { quoteCount += 1; quoteCents += Math.round(quote.value * 100); }
    if (payment.valid && payment.value !== null) { paymentCount += 1; paymentCents += Math.round(payment.value * 100); }
    if (receivable.valid && receivable.value !== null && payment.valid) {
      outstandingKnown += 1;
      outstandingCents += Math.max(Math.round(receivable.value * 100) - Math.round((payment.value || 0) * 100), 0);
    } else if (receivable.value !== null) outstandingUnknown += 1;
  });
  return { quoteCents, paymentCents, outstandingCents, quoteCount, paymentCount, outstandingKnown, outstandingUnknown };
}

function dimensionState(fact, judgment) {
  const hasFact = Boolean(cleanText(fact));

  if (!hasFact && !hasJudgment) return { label: "未复盘", width: 12, tone: "muted", summary: "本周还没有完成复盘" };

  if (!hasFact && hasJudgment) return { label: "需要补充", width: 34, tone: "danger", summary: "已经有结论，还需要补充依据" };

  return warning
    ? { label: "已判断 · 需关注", width: 76, tone: "gold", summary: cleanText(judgment) }
    : { label: "已完成判断", width: 92, tone: "teal", summary: cleanText(judgment) };
}

function buildAlerts() {
  const alerts = [];
  activeProjects().forEach((project) => {
    const quote = storedMoney(project.quote);
    const payment = storedMoney(project.payment);
    if (!project.next) alerts.push({ tone: "warning", title: "缺少唯一下一步", detail: project.client + " · " + project.project });
    if (!quote.valid || !payment.valid) alerts.push({ tone: "danger", title: "金额格式异常", detail: project.client + " · " + project.project });
    if (payment.valid && payment.value !== null && quote.valid && quote.value === null) alerts.push({ tone: "warning", title: "已有回款但报价为空", detail: project.client + " · " + project.project });
    if (quote.valid && quote.value !== null && payment.valid && payment.value !== null && payment.value > quote.value) alerts.push({ tone: "warning", title: "回款高于报价，需核账", detail: project.client + " · " + project.project });
  });
  if (state.calculation) {
    const fm = state.calculation.fm;
    const cash = finiteNumber(fm.cash_net_change);
    const safety = finiteNumber(fm.safety_value);
    const forecast = String(fm.calculation_mode || "") === "forecast";
    if (cash !== null && cash < 0) alerts.push({ tone: "warning", title: "现金正在减少", detail: "请检查到账节奏、实际支出与个人取款" });
    if (!forecast && safety !== null && safety < 0) alerts.push({ tone: "danger", title: "履约资金不够", detail: "完成已收款工作后仍存在资金缺口" });
    if (forecast && safety !== null && safety > 0) alerts.push({ tone: "warning", title: "创业预估存在资金缺口", detail: financeMoneyText(safety) });
  } else if (confirmedProjects().some((project) => finiteNumber(project.payment) > 0)) {
    alerts.push({ tone: "warning", title: "已有到账但利润未核算", detail: "请先完成一次真实创业算账" });
  }
  return alerts;
}

function renderFreshness() {
  const dates = [];
  state.projects.forEach((item) => dates.push(item.updatedAt));
  state.abilities.forEach((item) => dates.push(item.updatedAt));
  if (state.review) dates.push(state.review.updatedAt);
  if (state.business) dates.push(state.business.updatedAt);
  if (state.calculation) dates.push(state.calculation.updatedAt);
  if (state.financeInsight) dates.push(state.financeInsight.updatedAt);
  if (state.targetPlan) dates.push(state.targetPlan.updatedAt);
  if (state.actionPlan) dates.push(state.actionPlan.updatedAt);
  const valid = dates.map((value) => new Date(value)).filter((date) => Number.isFinite(date.getTime())).sort((a, b) => b - a);
  const label = valid.length ? valid[0].toLocaleString("zh-CN", { month: "2-digit", day: "2-digit", hour: "2-digit", minute: "2-digit", hour12: false }) : "尚无经营数据";
  q("[data-last-updated]").textContent = label;
  const sync = q("[data-sync-feedback]");
  if (sync && !sync.classList.contains("is-emphasis")) sync.textContent = valid.length ? "已更新 · " + label : "等待第一条经营记录";
}

function copyText(text) {
  if (navigator.clipboard && navigator.clipboard.writeText) return navigator.clipboard.writeText(text);
  const input = document.createElement("textarea");
  input.value = text;
  input.style.position = "fixed";
  input.style.opacity = "0";
  document.body.appendChild(input);
  input.select();
  document.execCommand("copy");
  input.remove();
  return Promise.resolve();
}

async function fillCommand(leaf, command) {
  if (!leaf || !leaf.containerEl) return false;
  await new Promise((resolve) => setTimeout(resolve, 160));
  const input = leaf.containerEl.querySelector("textarea, [contenteditable='true']");
  if (!input) return false;
  const existing = input instanceof HTMLTextAreaElement || input instanceof HTMLInputElement ? input.value : input.textContent;
  if (String(existing || "").trim()) return false;
  input.focus();
  if (input instanceof HTMLTextAreaElement || input instanceof HTMLInputElement) {
    const setter = Object.getOwnPropertyDescriptor(Object.getPrototypeOf(input), "value");
    if (setter && setter.set) setter.set.call(input, command); else input.value = command;
    input.dispatchEvent(new Event("input", { bubbles: true }));
  } else {
    input.textContent = command;
    input.dispatchEvent(new InputEvent("input", { bubbles: true, inputType: "insertText", data: command }));
  }
  return true;
}

function watchBusinessData() {
  let timer = null;
  const isBusinessPath = (path) => String(path || "").startsWith(PATHS.projects + "/")
    || String(path || "").startsWith(PATHS.archivedProjects + "/")
    || String(path || "").startsWith(PATHS.abilities + "/")
    || String(path || "").startsWith(PATHS.reviews + "/")
    || String(path || "").startsWith(PATHS.financeRoot + "/")
    || String(path || "").startsWith(PATHS.targetPlans + "/")
    || String(path || "").startsWith(PATHS.actionPlans + "/")
    || String(path || "") === PATHS.businessState;
  const schedule = (file) => {
    if (!file || !isBusinessPath(file.path)) return;
    clearTimeout(timer);
    timer = setTimeout(async () => {
      if (!ROOT.isConnected) return;
      try {
        await refresh();
        showSyncFeedback("经营数据已更新", "success");
        toast("经营数据已更新。", "success");
      } catch (error) {
        console.error("经营看板自动同步失败", error);
      }
    }, 260);
  };
  const refs = [
    app.vault.on("create", schedule),
    app.vault.on("modify", schedule),
    app.vault.on("delete", schedule),
    app.vault.on("rename", schedule)
  ];
  const observer = new MutationObserver(() => {
    if (ROOT.isConnected) return;
    clearTimeout(timer);
    refs.forEach((ref) => app.vault.offref(ref));
    observer.disconnect();
  });
  observer.observe(document.body, { childList: true, subtree: true });
}

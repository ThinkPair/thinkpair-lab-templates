function displayDate(value) {
  if (!value) return "日期待定";
  const parts = String(value).split("-").map(Number);
  if (parts.length !== 3 || parts.some((part) => !Number.isFinite(part))) return String(value);
  return parts[0] + "年" + parts[1] + "月" + parts[2] + "日";
}

function localDate(value) {
  const match = String(value || "").match(/^(\d{4})-(\d{2})-(\d{2})$/);
  if (!match) return null;
  const date = new Date(Number(match[1]), Number(match[2]) - 1, Number(match[3]));
  if (date.getFullYear() !== Number(match[1]) || date.getMonth() !== Number(match[2]) - 1 || date.getDate() !== Number(match[3])) return null;
  return date;
}

function fallbackPeriodProfile(inputOrStart, endValue) {
  const source = typeof inputOrStart === "object" && inputOrStart !== null ? inputOrStart : {};
  const startText = typeof inputOrStart === "string" ? inputOrStart : (source.periodStart || source.period_start);
  const endText = endValue || source.periodEnd || source.period_end;
  const start = localDate(startText);
  const end = localDate(endText);
  const errors = [];
  const warnings = [];
  if (!start) errors.push("开始日期无效");
  if (!end) errors.push("结束日期无效");
  if (start && end && end < start) errors.push("结束日期不能早于开始日期");
  if (errors.length) return { valid: false, errors, warnings, start: startText || "", end: endText || "", totalDays: null, monthFactor: null, isFullCalendarMonth: false, isLegacyPeriodAmbiguous: true };
  const totalDays = Math.round((end - start) / 86400000) + 1;
  let cursor = new Date(start.getFullYear(), start.getMonth(), 1);
  let monthFactor = 0;
  while (cursor <= end) {
    const monthStart = new Date(cursor.getFullYear(), cursor.getMonth(), 1);
    const monthEnd = new Date(cursor.getFullYear(), cursor.getMonth() + 1, 0);
    const coveredStart = start > monthStart ? start : monthStart;
    const coveredEnd = end < monthEnd ? end : monthEnd;
    if (coveredStart <= coveredEnd) monthFactor += (Math.round((coveredEnd - coveredStart) / 86400000) + 1) / monthEnd.getDate();
    cursor = new Date(cursor.getFullYear(), cursor.getMonth() + 1, 1);
  }
  const isFullCalendarMonth = start.getDate() === 1 && end.getFullYear() === start.getFullYear() && end.getMonth() === start.getMonth() && end.getDate() === new Date(start.getFullYear(), start.getMonth() + 1, 0).getDate();
  const oneYearPlusOneDay = start.getMonth() === 0 && start.getDate() === 1 && end.getFullYear() === start.getFullYear() + 1 && end.getMonth() === 0 && end.getDate() === 1;
  if (oneYearPlusOneDay) warnings.push("结束日期包含当天；若核算" + start.getFullYear() + "全年，请改为" + start.getFullYear() + "-12-31。");
  return { valid: true, errors, warnings, start: isoDate(start), end: isoDate(end), totalDays, monthFactor, periodMonthFactor: monthFactor, isFullCalendarMonth, isLegacyPeriodAmbiguous: !isFullCalendarMonth };
}

function friendlyPeriodLabel(startValue, endValue) {
  const start = localDate(startValue);
  const end = localDate(endValue);
  if (!start || !end || end < start) return displayDate(startValue) + "—" + displayDate(endValue);
  const lastOfStartMonth = new Date(start.getFullYear(), start.getMonth() + 1, 0);
  if (start.getDate() === 1 && end.getFullYear() === start.getFullYear() && end.getMonth() === start.getMonth() && end.getDate() === lastOfStartMonth.getDate()) return start.getFullYear() + "年" + (start.getMonth() + 1) + "月";
  const quarterStart = start.getDate() === 1 && start.getMonth() % 3 === 0;
  const quarterEnd = new Date(start.getFullYear(), start.getMonth() + 3, 0);
  if (quarterStart && end.getTime() === quarterEnd.getTime()) return start.getFullYear() + "年第" + (Math.floor(start.getMonth() / 3) + 1) + "季度";
  if (start.getMonth() === 0 && start.getDate() === 1 && end.getFullYear() === start.getFullYear() && end.getMonth() === 11 && end.getDate() === 31) return start.getFullYear() + "年全年";
  return displayDate(startValue) + "—" + displayDate(endValue) + "（含首尾日期）";
}

function periodPresetValue() {
  const now = new Date();
  const lastMonthStart = new Date(now.getFullYear(), now.getMonth() - 1, 1);
  const lastMonthEnd = new Date(now.getFullYear(), now.getMonth(), 0);
  const currentQuarterStartMonth = Math.floor(now.getMonth() / 3) * 3;
  const lastQuarterEnd = new Date(now.getFullYear(), currentQuarterStartMonth, 0);
  const lastQuarterStart = new Date(lastQuarterEnd.getFullYear(), lastQuarterEnd.getMonth() - 2, 1);
  const lastYearStart = new Date(now.getFullYear() - 1, 0, 1);
  const lastYearEnd = new Date(now.getFullYear() - 1, 11, 31);
  const matches = (a, b) => state.periodStart === isoDate(a) && state.periodEnd === isoDate(b);
  if (matches(lastMonthStart, lastMonthEnd)) return "last-month";
  if (matches(lastQuarterStart, lastQuarterEnd)) return "last-quarter";
  if (matches(lastYearStart, lastYearEnd)) return "last-year";
  return "custom";
}

function applyPeriodPreset(preset) {
  const now = new Date();
  if (preset === "last-month") {
    state.periodStart = isoDate(new Date(now.getFullYear(), now.getMonth() - 1, 1));
    state.periodEnd = isoDate(new Date(now.getFullYear(), now.getMonth(), 0));
  } else if (preset === "last-quarter") {
    const currentQuarterStartMonth = Math.floor(now.getMonth() / 3) * 3;
    const end = new Date(now.getFullYear(), currentQuarterStartMonth, 0);
    state.periodStart = isoDate(new Date(end.getFullYear(), end.getMonth() - 2, 1));
    state.periodEnd = isoDate(end);
  } else if (preset === "last-year") {
    state.periodStart = isoDate(new Date(now.getFullYear() - 1, 0, 1));
    state.periodEnd = isoDate(new Date(now.getFullYear() - 1, 11, 31));
  }
}

function applyForecastMonth(value) {
  const match = String(value || "").match(/^(\d{4})-(\d{2})$/);

}

function initializeIcons(scope = ROOT) {

    if (typeof window !== "undefined" && typeof window.setIcon === "function") applyIcon = window.setIcon;

    applyIcon = null;

  scope.querySelectorAll("[data-help-icon]:not([data-icon-ready])").forEach((node) => {
    applyIcon(node, "info");
    node.dataset.iconReady = "true";
  });
  scope.querySelectorAll("[data-lucide]:not([data-icon-ready])").forEach((node) => {
    applyIcon(node, node.dataset.lucide);
    node.dataset.iconReady = "true";
  });
}

function toneFrom(result) {
  const rawProfit = result?.sustainableProfit;
  const rawCash = result?.cashNetChange;
  if (!finiteValue(rawProfit)) return "neutral";
  const profit = Number(rawProfit);
  const cash = finiteValue(rawCash) ? Number(rawCash) : null;
  if (state.mode === "forecast") {
    const rawRate = result?.capacityRate ?? result?.breakEvenCapacityRate;
    const rate = finiteValue(rawRate) ? Number(rawRate) : null;
    if ((finiteValue(result?.unitContribution) && Number(result.unitContribution) <= 0) || (rate !== null && rate > 100)) return "danger";
    if ((rate !== null && rate > 80) || profit < 0) return "warning";
    return "healthy";
  }
  if (cash === null) return profit >= 0 ? "healthy" : "danger";
  if (profit > 0 && cash > 0) return "healthy";
  if (profit < 0 && cash < 0) return "danger";
  return "warning";
}

function actionFor(result) {
  const unit = state.unitLabel || "份";
  const rawProfit = result?.sustainableProfit;
  const rawCash = result?.cashNetChange;
  const rawSafety = result?.safetyValue;
  const rawContribution = result?.unitContribution;
  const rawCapacityRate = result?.capacityRate ?? result?.breakEvenCapacityRate;
  const profit = finiteValue(rawProfit) ? Number(rawProfit) : null;
  const cash = finiteValue(rawCash) ? Number(rawCash) : null;
  const safety = finiteValue(rawSafety) ? Number(rawSafety) : null;
  const contribution = finiteValue(rawContribution) ? Number(rawContribution) : null;
  const capacityRate = finiteValue(rawCapacityRate) ? Number(rawCapacityRate) : null;
  if (contribution !== null && contribution <= 0) return "先调整售价或直接成本，做到每完成 1 " + unit + " 都能留下钱，再谈扩大销量。";
  if (state.mode === "actual") {
    if (safety !== null && safety < 0) return "先暂停新增非必要支出，把未完成承诺和可用现金逐项核准，补上交付缺口。";
    if (profit !== null && profit < 0) return "先围绕保本线只改一个变量：优先验证提价或降低直接成本，别同时扩新项目。";
    if (cash !== null && cash < 0) return "先查清本月最大的现金流出和未到账款，把回款节奏拉回正数。";
    if (profit !== null) return "保持当前交付方式，再连续记录一个月，确认这次盈利不是偶然波动。";
  } else {
    if (capacityRate !== null && capacityRate > 100) return "先别启动：现有产能永远达不到保本量，必须先提价、降固定成本或提高产能。";
    if (capacityRate !== null && capacityRate > 80) return "先用真实客户测试提价或降低固定成本，让保本量降到最大产能的 80% 以下。";
    if (profit !== null && profit < 0) return "先做一次最小付费验证，不追加重投入；重点验证现实销量能否越过保本线。";
    if (profit !== null) return "先用最小付费交付验证 1—3 个真实客户，再决定是否扩大固定投入。";
  }
  return "先补齐下一项关键数字，再决定经营动作。";
}

function hasCurrentSavedCalculation() {

function sourceTargetValues() {
  const quick = calculation?.state?.quick || state.quick || {};
  const mode = calculation?.mode || state.mode;
  const recognizedRevenue = finiteValue(quick.recognizedRevenue) ? Number(quick.recognizedRevenue) : null;
  const completedUnits = finiteValue(quick.completedUnits) && Number(quick.completedUnits) > 0 ? Number(quick.completedUnits) : null;
  const directCost = finiteValue(quick.directCost) ? Number(quick.directCost) : null;
  const unitPrice = finiteValue(quick.unitPrice) ? Number(quick.unitPrice) : null;
  const unitDirectCost = finiteValue(quick.unitDirectCost) ? Number(quick.unitDirectCost) : null;
  let contributionMarginRate = null;
  let unitRecognizedRevenue = null;
  let derivedUnitDirectCost = null;
  if (mode === "forecast") {
    unitRecognizedRevenue = unitPrice;
    derivedUnitDirectCost = unitDirectCost;
    if (unitPrice !== null && unitPrice > 0 && unitDirectCost !== null) contributionMarginRate = ((unitPrice - unitDirectCost) / unitPrice) * 100;
  } else {
    if (recognizedRevenue !== null && recognizedRevenue > 0 && directCost !== null) contributionMarginRate = ((recognizedRevenue - directCost) / recognizedRevenue) * 100;
    if (recognizedRevenue !== null && completedUnits !== null) unitRecognizedRevenue = recognizedRevenue / completedUnits;
    if (directCost !== null && completedUnits !== null) derivedUnitDirectCost = directCost / completedUnits;
  }
  return {
    fixedOperatingCost: quick.monthlyFixedCostExOwner,
    ownerWage: quick.monthlyOwnerWage,
    contributionMarginRate,
    unitRecognizedRevenue,
    unitDirectCost: derivedUnitDirectCost,
    maxStableDeliveryUnits: mode === "forecast" ? quick.capacityUnits : null,
    averageCashCollectedPerSale: null,
    futureRigidCashOutflow: mode === "actual" ? quick.next30RigidOutflow : null
  };
}

function syncTargetState() {
  if (!hasCurrentSavedCalculation()) return;
  if (targetState.sourceInputHash === calculation.inputHash) return;
  targetState = {
    sourceInputHash: calculation.inputHash,
    targetMonth: currentMonthValue(),
    targetProfit: null,
    targetCashIncrease: "0",
    averageCashCollectedPerSale: null,
    unitRecognizedRevenue: null,
    maxStableDeliveryUnits: null
  };
  targetCalculation = null;
  lastTargetSaved = null;
}

function buildTargetInput() {
  const source = sourceTargetValues();
  const unitRecognizedRevenue = finiteValue(source.unitRecognizedRevenue) ? source.unitRecognizedRevenue : targetState.unitRecognizedRevenue;
  let unitDirectCost = source.unitDirectCost;
  if (!finiteValue(unitDirectCost) && finiteValue(unitRecognizedRevenue) && finiteValue(source.contributionMarginRate)) {
    unitDirectCost = Number(unitRecognizedRevenue) * (1 - Number(source.contributionMarginRate) / 100);
  }
  return {
    calculationId: lastSaved?.calculationId || null,
    sourceInputHash: lastSaved?.inputHash || null,
    targetMonth: targetState.targetMonth,
    targetProfit: targetState.targetProfit,
    targetCashIncrease: isFilled(targetState.targetCashIncrease) ? targetState.targetCashIncrease : 0,
    fixedOperatingCost: source.fixedOperatingCost,
    ownerWage: source.ownerWage,
    contributionMarginRate: source.contributionMarginRate,
    unitRecognizedRevenue,
    unitDirectCost,
    maxStableDeliveryUnits: finiteValue(source.maxStableDeliveryUnits) ? source.maxStableDeliveryUnits : targetState.maxStableDeliveryUnits,
    averageCashCollectedPerSale: finiteValue(source.averageCashCollectedPerSale) ? source.averageCashCollectedPerSale : targetState.averageCashCollectedPerSale,
    futureRigidCashOutflow: source.futureRigidCashOutflow,
    currency: "CNY",
    salesUnit: "单",
    deliveryUnit: state.unitLabel || "份"
  };
}

async function ensureFolder(path) {
  const parts = path.split("/");
  let current = "";
  for (const part of parts) {
    current = current ? current + "/" + part : part;
    if (!app.vault.getAbstractFileByPath(current)) await app.vault.createFolder(current);
  }
}

function recalculateTargetForSave() {
  if (!hasCurrentSavedCalculation()) throw new Error("核算数据已经变化，请先重新保存核算。");
  if (typeof ENGINE.calculateBusinessTargetPlan !== "function") throw new Error("目标公式尚未加载。");
  const rawInput = buildTargetInput();
  const validation = typeof ENGINE.validateBusinessTargetPlan === "function" ? ENGINE.validateBusinessTargetPlan(rawInput) : null;
  if (validation && validation.errors?.length) throw new Error(targetErrorText(validation) || "目标输入没有通过校验。");
  const envelope = ENGINE.calculateBusinessTargetPlan(rawInput);
  if (!envelope?.savable) throw new Error(targetErrorText(envelope) || (targetMissingText(envelope) ? "还差：“" + targetMissingText(envelope) + "”。" : "当前不能保存完整目标。"));

  const result = envelope.result || envelope.preview || {};
  const sourceInputHash = pickValue(envelope, "sourceInputHash", "source_input_hash") || input.sourceInputHash || input.source_input_hash;
  const targetInputHash = pickValue(envelope, "targetInputHash", "target_input_hash", "inputHash", "input_hash");
  const targetPlanId = pickValue(envelope, "targetPlanId", "target_plan_id") || (targetInputHash ? "btp-" + targetInputHash : null);

  if (!targetInputHash || !targetPlanId || !/^[A-Za-z0-9_-]+$/.test(String(targetPlanId))) throw new Error("目标标识生成失败，请重试。");
  return { envelope, input, result, sourceInputHash, targetInputHash, targetPlanId: String(targetPlanId) };
}

async function confirmSave() {
  if (saving) return;
  calculation = ENGINE.calculate(state);
  const profile = calculation.periodProfile || periodProfileFor(state);
  if (!profile?.valid) {
    q("[data-save-modal]").hidden = true;
    toast((profile?.errors || ["核算期间无效。"]).join("；"), "warning");
    return;
  }
  if (calculation.dataStatus !== "ready") {
    q("[data-save-modal]").hidden = true;
    toast("数据已经变化，请补齐后重新保存。", "warning");
    return;
  }
  if (lastSaved && lastSaved.inputHash === calculation.inputHash) {
    q("[data-save-modal]").hidden = true;
    toast("这组数字已经保存，无需重复创建。", "success");
    return;
  }
  saving = true;
  try {
    const calculationId = randomId(state.mode === "actual" ? "calc-actual" : "calc-forecast");
    const directory = state.mode === "actual" ? PATHS.actual : PATHS.forecast;
    await ensureFolder(directory);
    await ensureFolder(PATHS.plans);
    const recordPath = directory + "/" + calculationId + ".md";
    const planPath = PATHS.plans + "/plans-" + calculationId + ".md";
    await app.vault.create(recordPath, calculationRecordContent(calculationId, calculation, lastSaved?.calculationId || null));
    try { await app.vault.create(planPath, planRecordContent(calculationId, calculation)); } catch (error) { console.warn("收费方案索引未生成。", error); }
    const savedFile = app.vault.getAbstractFileByPath(recordPath);
    const verified = extractPayload(await app.vault.cachedRead(savedFile));
    if (verified.inputHash !== calculation.inputHash) throw new Error("保存校验没有通过，请重试。");
    try { await consumeActiveFinanceDraft(calculationId); } catch (draftError) { console.warn("正式记录已保存，但草稿状态未更新。", draftError); }
    lastSaved = { calculationId, inputHash: calculation.inputHash, path: recordPath, mode: state.mode, savedAt: nowIso() };
    localStorage.removeItem(DRAFT_KEY);
    q("[data-draft-status]").textContent = "本次结果已保存";
    q("[data-save-modal]").hidden = true;
    toast("核算已保存，首页会自动读取这份结果。", "success");
    await renderHistory();
    renderResults();
  } catch (error) {
    toast("保存失败：" + error.message, "warning");
  } finally {
    saving = false;
  }
}

async function listCalculationFiles() {
  const files = [];
  for (const directory of [PATHS.actual, PATHS.forecast]) {
    const folder = app.vault.getAbstractFileByPath(directory);
    if (!folder?.children) continue;
    folder.children.filter((file) => file.extension === "md" && file.name !== "README").forEach((file) => files.push(file));
  }
  return files.map((file) => ({ file, fm: app.metadataCache.getFileCache(file)?.frontmatter || {} }))
    .filter((item) => item.fm.record_type === "business_calculation" && item.fm.calculation_status === "confirmed" && String(item.fm.record_status || "saved") === "saved")
    .sort((a, b) => String(b.fm.updated_at || b.fm.created_at || "").localeCompare(String(a.fm.updated_at || a.fm.created_at || "")));
}

function legacyRecordNeedsReview(fm, payload = null) {

  const source = payload?.state || { inputMode: "quick_v2", periodStart: fm?.period_start, periodEnd: fm?.period_end };
  const profile = legacyPeriodProfile(source);
  return Boolean(profile?.isLegacyPeriodAmbiguous ?? !profile?.isFullCalendarMonth);
}

function convertLegacyState(rawState, basis, profile) {
  const mode = rawState?.mode === "forecast" ? "forecast" : "actual";
  const converted = blankQuick(mode, { offerName: rawState?.offerName || "", unitLabel: rawState?.unitLabel || "单", deliveryModel: rawState?.deliveryModel || "instant" });
  converted.periodStart = rawState?.periodStart || converted.periodStart;
  converted.periodEnd = rawState?.periodEnd || converted.periodEnd;
  const oldQuick = rawState?.quick || {};
  const allowed = mode === "actual"
    ? ["recognizedRevenue", "completedUnits", "cashIn", "cashOut", "directCost", "availableCash", "remainingFulfillmentCost", "next30RigidOutflow"]
    : ["unitPrice", "unitDirectCost", "baseUnits", "capacityUnits", "cashAfterStartup", "startupInvestment"];
  allowed.forEach((key) => { if (oldQuick[key] !== undefined) converted.quick[key] = oldQuick[key]; });
  const factor = Number(profile?.monthFactor ?? profile?.periodMonthFactor);
  const normalize = (value) => {
    if (!finiteValue(value)) return value ?? null;
    if (basis !== "period-total") return value;
    if (!Number.isFinite(factor) || factor <= 0) throw new Error("旧记录的核算月数无法确定。");
    return Math.round(Number(value) / factor * 100) / 100;
  };
  converted.quick.monthlyFixedCostExOwner = normalize(oldQuick.fixedCostExOwner ?? oldQuick.monthlyFixedCostExOwner);

function extractFinanceDraftPayload(text) {
  const match = String(text || "").match(/<!-- finance-draft-json:start -->\s*(?:```json\s*)?([\s\S]*?)(?:\s*```)?\s*<!-- finance-draft-json:end -->/);
  if (!match) throw new Error("核算草稿缺少 finance-draft-json 数据块。");
  return JSON.parse(match[1].trim());
}

function sanitizeFinanceDraftState(raw) {
  const mode = raw?.mode === "forecast" ? "forecast" : "actual";
  const clean = blankQuick(mode, { offerName: String(raw?.offerName || ""), unitLabel: String(raw?.unitLabel || "单"), deliveryModel: String(raw?.deliveryModel || "instant") });
  clean.periodStart = String(raw?.periodStart || clean.periodStart);
  clean.periodEnd = String(raw?.periodEnd || clean.periodEnd);
  const allowed = mode === "actual"
    ? ["recognizedRevenue", "completedUnits", "cashIn", "cashOut", "directCost", "monthlyFixedCostExOwner", "monthlyOwnerWage", "availableCash", "remainingFulfillmentCost", "next30RigidOutflow"]
    : ["unitPrice", "unitDirectCost", "monthlyFixedCostExOwner", "monthlyOwnerWage", "baseUnits", "capacityUnits", "cashAfterStartup", "startupInvestment"];
  allowed.forEach((key) => {
    const value = raw?.quick?.[key];
    if (value === null || value === undefined || value === "") clean.quick[key] = null;
    else if (finiteValue(value) && Number(value) >= 0) clean.quick[key] = value;
    else throw new Error("草稿字段“" + key + "”不是有效的非负数字。");
  });
  return ENGINE.cleanQuickState(clean);
}

async function copyText(text) {
  if (navigator.clipboard?.writeText) return navigator.clipboard.writeText(text);
  const input = document.createElement("textarea");
  input.value = text;
  input.style.position = "fixed";

}

async function fillAI(leaf, prompt) {
  await new Promise((resolve) => setTimeout(resolve, 160));
  const input = leaf?.containerEl?.querySelector("textarea, [contenteditable='true']");
  if (!input || String(input.value || input.textContent || "").trim()) return false;
  input.focus();
  if (input instanceof HTMLTextAreaElement || input instanceof HTMLInputElement) {
    const setter = Object.getOwnPropertyDescriptor(Object.getPrototypeOf(input), "value");
    if (setter?.set) setter.set.call(input, prompt); else input.value = prompt;
    input.dispatchEvent(new Event("input", { bubbles: true }));
  } else {
    input.textContent = prompt;
    input.dispatchEvent(new InputEvent("input", { bubbles: true, inputType: "insertText", data: prompt }));
  }
  return true;
}

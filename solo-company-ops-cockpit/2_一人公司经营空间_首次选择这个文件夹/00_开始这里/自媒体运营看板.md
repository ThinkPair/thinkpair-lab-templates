function escapeHtml(value) {
  return String(value ?? "")
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;")
    .replaceAll("'", "&#039;");
}

function toNumber(value) {
  if (value === null || value === undefined || value === "") return null;
  const number = Number(value);
  return Number.isFinite(number) ? number : null;
}

function toDate(value) {
  if (!value) return null;
  try {
    if (typeof value.toJSDate === "function") return value.toJSDate();
    if (value instanceof Date) return value;
    const parsed = new Date(String(value));
    return Number.isNaN(parsed.getTime()) ? null : parsed;
  } catch (error) {
    return null;
  }
}

function formatNumber(value) {
  if (value === null || value === undefined) return "—";
  return new Intl.NumberFormat("zh-CN", { maximumFractionDigits: 1 }).format(value);
}

function formatCurrency(value) {
  if (value === null || value === undefined) return "—";
  return `¥${new Intl.NumberFormat("zh-CN", { maximumFractionDigits: 0 }).format(value)}`;
}

function formatDateTime(value) {
  const date = toDate(value);
  if (!date) return "尚未更新";
  return new Intl.DateTimeFormat("zh-CN", {
    month: "2-digit",
    day: "2-digit",
    hour: "2-digit",
    minute: "2-digit",
    hour12: false
  }).format(date).replaceAll("/", "-");
}

function formatShortDate(value) {
  const date = toDate(value);
  if (!date) return "待补";
  return new Intl.DateTimeFormat("zh-CN", {
    month: "2-digit",
    day: "2-digit"
  }).format(date).replaceAll("/", "-");
}

function latestDate() {
  const candidates = [
    toDate(config.updated_at),
    ...snapshotPages.map((page) => toDate(page.snapshot_at)),
    ...eventPages.map((page) => toDate(page.event_at))
  ].filter(Boolean);
  return candidates.length ? new Date(Math.max(...candidates.map((date) => date.getTime()))) : new Date();
}

function rangeStart(anchor, range) {
  const result = new Date(anchor);
  result.setDate(result.getDate() - (range === "12w" ? 83 : 27));
  result.setHours(0, 0, 0, 0);
  return result;
}

function inRange(value, start, end) {
  const date = toDate(value);
  if (!date) return false;
  return date >= start && date <= end;
}

function stateLabel(value) {

}

}

function applyIcons(scope = ROOT) {
  let iconSetter = null;
  try {
    if (typeof window.setIcon === "function") iconSetter = window.setIcon;
    else if (typeof require === "function") iconSetter = require("obsidian").setIcon;
  } catch (error) {
    iconSetter = null;
  }
  if (!iconSetter) return;
  scope.querySelectorAll("[data-sm-icon]").forEach((node) => {
    if (node.dataset.iconReady === node.dataset.smIcon) return;
    try {
      node.replaceChildren();
      iconSetter(node, node.dataset.smIcon);
      node.dataset.iconReady = node.dataset.smIcon;
    } catch (error) {
      console.debug("图标加载失败，不影响看板使用。", error);
    }
  });
}

function mergeSnapshots(rows) {
  const groups = new Map();
  rows
    .slice()
    .sort((a, b) => (toDate(a.snapshot_at)?.getTime() || 0) - (toDate(b.snapshot_at)?.getTime() || 0))
    .forEach((page) => {
      const contentId = String(page.content_id || page.file?.name || "unknown");
      const platform = String(page.platform || "unknown");
      const key = `${contentId}::${platform}`;
      const current = groups.get(key) || {
        key,
        contentId,
        title: page.title || contentId,
        platform,
        publishedAt: null,
        snapshotAt: null,
        factState: "unknown",
        sourceRef: null,
        unknownFields: new Set(),
        metrics: {},
        metricDates: {}
      };
      current.title = page.title || current.title;
      current.publishedAt = toDate(page.published_at) || current.publishedAt;
      current.snapshotAt = toDate(page.snapshot_at) || current.snapshotAt;
      current.factState = page.fact_state || current.factState;
      current.sourceRef = page.source_ref || current.sourceRef;
      String(page.unknown_fields || "")
        .split(/[、,，/]/)
        .map((item) => item.trim())
        .filter(Boolean)
        .forEach((item) => current.unknownFields.add(item));
      SNAPSHOT_FIELDS.forEach((field) => {
        const value = toNumber(page[field]);
        if (value !== null) {
          current.metrics[field] = value;
          current.metricDates[field] = toDate(page.snapshot_at);
        }
      });
      groups.set(key, current);
    });
  return Array.from(groups.values());
}

function sumSnapshotField(rows, field, platforms) {
  const values = {};
  platforms.forEach((platform) => {
    const knownRows = rows.filter((row) => row.platform === platform && row.metrics[field] !== undefined);
    values[platform] = knownRows.length
      ? knownRows.reduce((sum, row) => sum + row.metrics[field], 0)
      : null;
  });
  return values;
}

function sumEventType(rows, type, platforms) {
  const values = {};
  platforms.forEach((platform) => {
    const matching = rows.filter((row) => String(row.platform) === platform && String(row.event_type) === type);
    values[platform] = matching.length
      ? matching.reduce((sum, row) => sum + (toNumber(row.count) ?? 0), 0)
      : null;
  });
  return values;
}

function sumRevenue(rows, platforms) {
  const values = {};
  platforms.forEach((platform) => {
    const matching = rows.filter((row) =>
      String(row.platform) === platform &&
      ["first_paid", "renewal"].includes(String(row.event_type)) &&
      toNumber(row.amount) !== null
    );
    values[platform] = matching.length
      ? matching.reduce((sum, row) => sum + (toNumber(row.amount) ?? 0), 0)
      : null;
  });
  return values;
}

function visibleValues(values) {
  if (smState.platform === "all") return values;
  return { [smState.platform]: values[smState.platform] ?? null };
}

function summarizeValues(values) {
  const entries = Object.entries(visibleValues(values));
  const known = entries.filter(([, value]) => value !== null && value !== undefined);
  return {
    value: known.length ? known.reduce((sum, [, value]) => sum + value, 0) : null,
    knownCount: known.length,
    expectedCount: entries.length,
    partial: known.length > 0 && known.length < entries.length
  };
}

function buildStage(id, label, unit, unitType, values, evidence) {
  const summary = summarizeValues(values);
  return {
    id,
    label,
    unit,
    unitType,
    values,
    evidence,
    ...summary
  };
}

function weekStart(date) {
  const result = new Date(date);
  result.setHours(0, 0, 0, 0);
  const day = result.getDay() || 7;
  result.setDate(result.getDate() - day + 1);
  return result;
}

function buildWeeks(anchor, count) {
  const endWeek = weekStart(anchor);
  return Array.from({ length: count }, (_, index) => {
    const start = new Date(endWeek);
    start.setDate(endWeek.getDate() - (count - 1 - index) * 7);
    const end = new Date(start);
    end.setDate(start.getDate() + 6);
    return {
      key: start.toISOString().slice(0, 10),
      start,
      end,
      label: `${start.getMonth() + 1}/${start.getDate()}—${end.getMonth() + 1}/${end.getDate()}`,
      values: Object.fromEntries(enabledPlatforms.map((platform) => [platform, null]))
    };
  });
}

  function drawDonut() {
    const canvas = ROOT.querySelector('[data-chart="donut"]');
    if (!canvas) return;
    const rect = canvas.getBoundingClientRect();
    if (!rect.width || !rect.height) return;
    const scale = Math.min(window.devicePixelRatio || 1, 2);
    canvas.width = Math.round(rect.width * scale);
    canvas.height = Math.round(rect.height * scale);
    const ctx = canvas.getContext("2d");
    ctx.scale(scale, scale);
    const width = rect.width;
    const height = rect.height;
    const cx = width / 2;
    const cy = height / 2;
    const radius = Math.min(width, height) * 0.34;
    const lineWidth = Math.max(16, radius * 0.28);
    ctx.clearRect(0, 0, width, height);
    ctx.lineCap = "round";
    ctx.lineWidth = lineWidth;
    ctx.strokeStyle = "rgba(82,75,101,.09)";
    ctx.beginPath();
    ctx.arc(cx, cy, radius, 0, Math.PI * 2);
    ctx.stroke();
    if (stages[2].value === null || stages[2].value <= 0) return;
    let angle = -Math.PI / 2;
    activePlatforms.forEach((platform) => {
      const value = leadValues[platform];
      if (value === null || value <= 0) return;
      const slice = (value / stages[2].value) * Math.PI * 2;
      const gap = Math.min(0.045, slice * 0.08);
      ctx.strokeStyle = PLATFORM_META[platform].color;
      ctx.globalAlpha = smState.platform === "all" || smState.platform === platform ? 0.92 : 0.2;
      ctx.beginPath();
      ctx.arc(cx, cy, radius, angle + gap, angle + slice - gap);
      ctx.stroke();
      angle += slice;
    });
    ctx.globalAlpha = 1;
  }

  function drawTrend() {
    const canvas = ROOT.querySelector('[data-chart="trend"]');
    if (!canvas) return;
    const rect = canvas.getBoundingClientRect();
    if (!rect.width || !rect.height) return;
    const scale = Math.min(window.devicePixelRatio || 1, 2);
    canvas.width = Math.round(rect.width * scale);
    canvas.height = Math.round(rect.height * scale);
    const ctx = canvas.getContext("2d");
    ctx.scale(scale, scale);
    const width = rect.width;
    const height = rect.height;
    const pad = { left: 24, right: 18, top: 18, bottom: 28 };
    const chartW = width - pad.left - pad.right;
    const chartH = height - pad.top - pad.bottom;
    ctx.clearRect(0, 0, width, height);
    ctx.font = '10px "Segoe UI", sans-serif';
    ctx.fillStyle = "#8B8591";
    ctx.textAlign = "center";
    ctx.lineWidth = 1;
    for (let index = 0; index < 4; index += 1) {
      const y = pad.top + (chartH / 3) * index;
      ctx.strokeStyle = "rgba(57,51,75,.08)";
      ctx.setLineDash([2, 5]);
      ctx.beginPath();
      ctx.moveTo(pad.left, y);
      ctx.lineTo(width - pad.right, y);
      ctx.stroke();
    }
    ctx.setLineDash([]);
    const weeks = buildWeeks(anchor, smState.range === "12w" ? 12 : 4);
    periodEvents
      .filter((page) => page.event_type === "qualified_lead")
      .forEach((page) => {
        const eventDate = toDate(page.event_at);
        if (!eventDate) return;
        const key = weekStart(eventDate).toISOString().slice(0, 10);
        const week = weeks.find((item) => item.key === key);
        const platform = String(page.platform || "");
        if (!week || !activePlatforms.includes(platform)) return;
        week.values[platform] = (week.values[platform] ?? 0) + (toNumber(page.count) ?? 0);
      });
    const shown = smState.platform === "all" ? activePlatforms : [smState.platform];
    const allValues = weeks.flatMap((week) => shown.map((platform) => week.values[platform]).filter((value) => value !== null));
    weeks.forEach((week, index) => {
      if (smState.range === "4w" || index % 2 === 0 || index === weeks.length - 1) {
        const x = pad.left + (chartW * index) / Math.max(1, weeks.length - 1);
        ctx.fillText(week.label, x, height - 7);
      }
    });
    if (!allValues.length) {
      ctx.fillStyle = "#8C8792";
      ctx.font = '12px "Microsoft YaHei", sans-serif';
      ctx.fillText("暂无有效线索趋势", width / 2, pad.top + chartH / 2);
      return;
    }
    const maxValue = Math.max(1, ...allValues) * 1.16;
    shown.forEach((platform, platformIndex) => {
      const points = weeks.map((week, index) => {
        const value = week.values[platform];
        if (value === null) return null;
        return {
          x: pad.left + (chartW * index) / Math.max(1, weeks.length - 1),
          y: pad.top + chartH - (value / maxValue) * chartH,
          value
        };
      });
      ctx.strokeStyle = PLATFORM_META[platform]?.color || "#6D4CFF";
      ctx.fillStyle = PLATFORM_META[platform]?.color || "#6D4CFF";
      ctx.globalAlpha = platformIndex === 0 ? 1 : 0.72;
      ctx.lineWidth = platformIndex === 0 ? 2.6 : 1.8;
      ctx.beginPath();
      let drawing = false;
      points.forEach((point) => {
        if (!point) {
          drawing = false;
          return;
        }
        if (!drawing) {
          ctx.moveTo(point.x, point.y);
          drawing = true;
        } else {
          ctx.lineTo(point.x, point.y);
        }
      });
      ctx.stroke();
      points.filter(Boolean).forEach((point) => {
        ctx.beginPath();
        ctx.arc(point.x, point.y, platformIndex === 0 ? 4 : 3, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = "#FFFFFF";
        ctx.beginPath();
        ctx.arc(point.x, point.y, platformIndex === 0 ? 1.7 : 1.2, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = PLATFORM_META[platform]?.color || "#6D4CFF";
      });
      ctx.globalAlpha = 1;
    });
  }
